function enviarOcrAN8N() { const WEBHOOK_OCR_URL = 'https://n8n.srv1233440.hstgr.cloud/webhook-test/af6da9c5-01e4-44f5-9fe6-7854016aa065'; const datos = { initData: Telegram.WebApp.initData, imagenBase64: imagenBase64 }; fetch(WEBHOOK_OCR_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(datos) }) .then(response => { // Si n8n devuelve un error 500 o 404, lo atrapamos aquí if (!response.ok) { return response.text().then(text => { throw new Error("Error de n8n (Código " + response.status + "): " + text); }); } return response.text(); }) .then(texto => { // 1. Mostramos el texto exacto que devolvió n8n // Telegram.WebApp.showAlert("Respuesta de n8n: " + texto); let res; try { const parsed = JSON.parse(texto); res = Array.isArray(parsed) ? parsed[0] : parsed; } catch (e) { // Si n8n devolvió HTML o texto en lugar de JSON, saltará aquí throw new Error("n8n no devolvió un JSON válido. Recibido: " + texto); } // 2. Llenamos los datos document.getElementById('tipoDoc').value = res.tipoDoc || 'Boleta'; document.getElementById('monto').value = res.monto || 0; document.getElementById('fecha').value = res.fecha || new Date().toISOString().split('T')[0]; // 3. Llenamos las obras const selectObra = document.getElementById('obra'); selectObra.innerHTML = ''; if (res.data && Array.isArray(res.data)) { res.data.forEach(item => { const option = document.createElement('option'); option.value = item.OBRA; option.text = item.OBRA; selectObra.appendChild(option); }); } else { selectObra.innerHTML = ''; } // 4. Pasamos a la pantalla 2 document.getElementById('pantallaCarga').classList.add('hidden'); document.getElementById('pantalla2').classList.remove('hidden'); }) .catch(err => { // AQUÍ MOSTRAMOS EL ERROR REAL EN PANTALLA console.error(err); Telegram.WebApp.showAlert("⚠️ Fallo la comunicación:\n\n" + err.message); document.getElementById('pantallaCarga').classList.add('hidden'); document.getElementById('pantalla2').classList.remove('hidden'); // Lo dejamos pasar a mano }); }
